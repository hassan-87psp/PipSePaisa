import "jsr:@supabase/functions-js/edge-runtime.d.ts";
import { createClient } from "jsr:@supabase/supabase-js@2";
import nodemailer from "npm:nodemailer@6.10.1";

const SMTP_HOST = Deno.env.get("SMTP_HOST") ?? "";
const SMTP_PORT = Number(Deno.env.get("SMTP_PORT") ?? "587");
const SMTP_USERNAME = Deno.env.get("SMTP_USERNAME") ?? "";
const SMTP_PASSWORD = Deno.env.get("SMTP_PASSWORD") ?? "";
const SMTP_FROM_EMAIL = Deno.env.get("SMTP_FROM_EMAIL") ?? "no-reply@pipsepaisa.com";
const SMTP_FROM_NAME = Deno.env.get("SMTP_FROM_NAME") ?? "PipSePaisa";
const SUPABASE_URL = Deno.env.get("SUPABASE_URL") ?? "";
const SUPABASE_SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? "";
const SITE_URL = "https://www.pipsepaisa.com";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
};

type EmailType =
  | "free_course_enrolled"
  | "payment_received"
  | "payment_approved"
  | "payment_rejected"
  | "payment_revoked"
  | "pin_access_welcome";

type Body = {
  type: EmailType;
  user_name?: string;
  course_title?: string;
  amount?: string | number;
  payment_method?: string;
  transaction_id?: string;
  rejection_reason?: string;
  target_user_id?: string;
};

function esc(value: unknown): string {
  return String(value ?? "")
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}

function json(data: Record<string, unknown>, status = 200): Response {
  return new Response(JSON.stringify(data), {
    status,
    headers: { ...corsHeaders, "Content-Type": "application/json" },
  });
}

function button(label: string, url: string): string {
  return `<p style="margin:26px 0 8px"><a href="${esc(url)}" style="display:inline-block;background:#f59e0b;color:#111827;text-decoration:none;padding:14px 22px;border-radius:10px;font-weight:800">${esc(label)}</a></p>`;
}

function layout(title: string, content: string): string {
  return `<!doctype html><html><body style="margin:0;background:#f3f4f6;font-family:Arial,Helvetica,sans-serif;color:#111827"><table width="100%" cellpadding="0" cellspacing="0" style="padding:30px 12px"><tr><td align="center"><table width="100%" cellpadding="0" cellspacing="0" style="max-width:620px;background:#fff;border-radius:16px;overflow:hidden;border:1px solid #e5e7eb"><tr><td style="background:#0b172a;padding:24px;text-align:center"><div style="font-size:25px;font-weight:900;color:#fff">Pip<span style="color:#f59e0b">Se</span>Paisa</div><div style="font-size:12px;color:#cbd5e1;margin-top:5px">Grow With Us</div></td></tr><tr><td style="padding:32px 28px"><h1 style="font-size:24px;margin:0 0 18px">${esc(title)}</h1>${content}</td></tr><tr><td style="padding:18px 28px;background:#fafafa;border-top:1px solid #e5e7eb;color:#6b7280;font-size:12px;line-height:1.6">This is an automated email from PipSePaisa. Please do not reply.<br>© ${new Date().getFullYear()} PipSePaisa · www.pipsepaisa.com</td></tr></table></td></tr></table></body></html>`;
}

function formatDeadline(value: string | null | undefined): string {
  if (!value) return "the deadline shown in Settings";
  try {
    return new Date(value).toLocaleString("en-GB", {
      timeZone: "Asia/Kuala_Lumpur",
      dateStyle: "medium",
      timeStyle: "short",
    }) + " (Malaysia time)";
  } catch {
    return "the deadline shown in Settings";
  }
}

function buildEmail(body: Body, extra: Record<string, unknown> = {}) {
  const name = esc(body.user_name || "Student");
  const course = esc(body.course_title || "PipSePaisa Forex Course");
  const amount = esc(body.amount || "Not provided");
  const method = esc(body.payment_method || "Not provided");
  const txn = esc(body.transaction_id || "Not provided");
  const reason = esc(body.rejection_reason || "The payment details could not be verified.");

  if (body.type === "free_course_enrolled") {
    return {
      subject: `Enrollment Successful — ${course}`,
      html: layout("Course Enrollment Successful", `<p>Hi <strong>${name}</strong>,</p><p style="line-height:1.7;color:#374151">Your enrollment in <strong>${course}</strong> has been completed successfully.</p><div style="padding:16px;background:#ecfdf5;border:1px solid #a7f3d0;border-radius:10px;color:#065f46"><strong>✓ Enrollment confirmed</strong><br><strong>✓ Course access active</strong></div>${button("Open My Course", SITE_URL)}`),
    };
  }

  if (body.type === "payment_received") {
    return {
      subject: "Payment Received — Under Review",
      html: layout("Payment Received", `<p>Hi <strong>${name}</strong>,</p><p style="line-height:1.7;color:#374151">We received your payment details for <strong>${course}</strong>.</p><table width="100%" style="background:#f9fafb;border:1px solid #e5e7eb;border-radius:10px"><tr><td style="padding:11px">Amount</td><td style="padding:11px;font-weight:700">${amount}</td></tr><tr><td style="padding:11px">Payment Method</td><td style="padding:11px;font-weight:700">${method}</td></tr><tr><td style="padding:11px">Transaction ID</td><td style="padding:11px;font-weight:700">${txn}</td></tr><tr><td style="padding:11px">Status</td><td style="padding:11px;font-weight:700;color:#d97706">Under Review</td></tr></table><div style="margin-top:18px;padding:15px;background:#fff7ed;border:1px solid #fed7aa;border-radius:10px;color:#9a3412">Your paid course remains locked until admin approval.</div>${button("View Payment Status", SITE_URL)}`),
    };
  }

  if (body.type === "payment_approved") {
    return {
      subject: "Payment Approved — Course Unlocked",
      html: layout("Payment Approved", `<p>Hi <strong>${name}</strong>,</p><p style="line-height:1.7;color:#374151">Your payment for <strong>${course}</strong> has been verified and approved.</p><div style="padding:16px;background:#ecfdf5;border:1px solid #a7f3d0;border-radius:10px;color:#065f46"><strong>✓ Payment approved</strong><br><strong>✓ Enrollment active</strong><br><strong>✓ Course modules unlocked</strong></div>${button("Open My Course", SITE_URL)}`),
    };
  }

  if (body.type === "payment_rejected" || body.type === "payment_revoked") {
    const revoked = body.type === "payment_revoked";
    return {
      subject: revoked ? "Course Access Revoked — Payment Review Required" : "Payment Verification Required",
      html: layout(revoked ? "Course Access Revoked" : "Payment Could Not Be Verified", `<p>Hi <strong>${name}</strong>,</p><p style="line-height:1.7;color:#374151">${revoked ? "Your previously approved course access has been revoked for review." : `We could not verify your submitted payment for <strong>${course}</strong>.`}</p><div style="padding:16px;background:#fef2f2;border:1px solid #fecaca;border-radius:10px;color:#991b1b"><strong>Reason:</strong><br>${reason}</div><p style="line-height:1.7;color:#374151">Your course is locked. Please contact the admin or resubmit the requested payment details.</p>${button("Open My Courses", SITE_URL)}`),
    };
  }

  const whatsapp = String(extra.admin_whatsapp || "601156558689").replace(/\D/g, "");
  const deadline = formatDeadline(String(extra.grace_expires_at || ""));
  const graceLabel = esc(extra.grace_label || "48 hours");
  return {
    subject: "Your Free PipSePaisa Access PIN",
    html: layout("Activate Your Free Access PIN", `<p>Hi <strong>${name}</strong>,</p><p style="line-height:1.7;color:#374151">Your PipSePaisa account has been verified successfully.</p><div style="padding:16px;background:#fff7ed;border:1px solid #fed7aa;border-radius:10px;color:#9a3412"><strong>Your access PIN is completely free.</strong><br>Contact the PipSePaisa admin on WhatsApp to request your unique PIN, then add it in <strong>Settings → Access PIN</strong>.</div><p style="line-height:1.7;color:#374151">You have <strong>${graceLabel}</strong> to add the PIN. Your current deadline is <strong>${esc(deadline)}</strong>. After the deadline, Signals, Charts, Articles, Journal and other protected features will remain visible but locked until the PIN is activated.</p>${button("Contact Admin on WhatsApp", `https://wa.me/${whatsapp}`)}${button("Open PipSePaisa Settings", `${SITE_URL}/index.html?open=settings`)}`),
  };
}

function validateConfig() {
  const missing = [
    ["SMTP_HOST", SMTP_HOST],
    ["SMTP_USERNAME", SMTP_USERNAME],
    ["SMTP_PASSWORD", SMTP_PASSWORD],
    ["SMTP_FROM_EMAIL", SMTP_FROM_EMAIL],
    ["SUPABASE_URL", SUPABASE_URL],
    ["SUPABASE_SERVICE_ROLE_KEY", SUPABASE_SERVICE_ROLE_KEY],
  ].filter(([, value]) => !value).map(([key]) => key);
  if (missing.length) throw new Error(`Missing environment variables: ${missing.join(", ")}`);
}

const transporter = nodemailer.createTransport({
  host: SMTP_HOST,
  port: SMTP_PORT,
  secure: SMTP_PORT === 465,
  requireTLS: SMTP_PORT === 587,
  auth: { user: SMTP_USERNAME, pass: SMTP_PASSWORD },
  connectionTimeout: 15000,
  greetingTimeout: 15000,
  socketTimeout: 25000,
  tls: { servername: SMTP_HOST, rejectUnauthorized: true },
});

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: corsHeaders });
  if (req.method !== "POST") return json({ success: false, error: "Only POST requests are allowed." }, 405);

  try {
    validateConfig();
    const authorization = req.headers.get("Authorization");
    if (!authorization?.startsWith("Bearer ")) return json({ success: false, error: "Authentication required." }, 401);
    const token = authorization.replace("Bearer ", "").trim();

    const admin = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, {
      auth: { persistSession: false, autoRefreshToken: false },
    });
    const { data: userResult, error: userError } = await admin.auth.getUser(token);
    const currentUser = userResult?.user;
    if (userError || !currentUser?.email) return json({ success: false, error: "Your login session is invalid or expired." }, 401);

    const body = await req.json() as Body;
    const allowed: EmailType[] = ["free_course_enrolled", "payment_received", "payment_approved", "payment_rejected", "payment_revoked", "pin_access_welcome"];
    if (!allowed.includes(body.type)) return json({ success: false, error: "Unsupported email type." }, 400);

    let recipientEmail = currentUser.email;
    let recipientName = body.user_name || currentUser.user_metadata?.full_name || currentUser.user_metadata?.name || "Student";
    const extra: Record<string, unknown> = {};

    if (["payment_approved", "payment_rejected", "payment_revoked"].includes(body.type)) {
      const { data: profile } = await admin.from("profiles").select("role").eq("id", currentUser.id).maybeSingle();
      const role = String(profile?.role || "").toLowerCase();
      if (!["admin", "owner", "super_admin", "superadmin"].includes(role)) return json({ success: false, error: "Admin access is required." }, 403);
      if (!body.target_user_id) return json({ success: false, error: "target_user_id is required." }, 400);
      const { data: targetResult, error: targetError } = await admin.auth.admin.getUserById(body.target_user_id);
      if (targetError || !targetResult?.user?.email) return json({ success: false, error: "Student account was not found." }, 404);
      recipientEmail = targetResult.user.email;
      recipientName = body.user_name || targetResult.user.user_metadata?.full_name || targetResult.user.user_metadata?.name || "Student";
    }

    if (body.type === "pin_access_welcome") {
      const [{ data: pin }, { data: settings }] = await Promise.all([
        admin.from("user_access_pins").select("grace_expires_at").eq("user_id", currentUser.id).maybeSingle(),
        admin.from("pin_access_settings").select("grace_value,grace_unit,admin_whatsapp").eq("id", 1).maybeSingle(),
      ]);
      extra.grace_expires_at = pin?.grace_expires_at || null;
      extra.admin_whatsapp = settings?.admin_whatsapp || "601156558689";
      extra.grace_label = `${settings?.grace_value || 48} ${settings?.grace_unit || "hours"}`;
    }

    const email = buildEmail({ ...body, user_name: recipientName }, extra);
    const result = await transporter.sendMail({
      from: { name: SMTP_FROM_NAME, address: SMTP_FROM_EMAIL },
      to: recipientEmail,
      subject: email.subject,
      html: email.html,
    });

    return json({ success: true, message: "Email sent successfully.", recipient: recipientEmail, message_id: result.messageId || null });
  } catch (error) {
    console.error("send-course-email error", error);
    return json({ success: false, error: error instanceof Error ? error.message : "Unexpected server error." }, 500);
  }
});
